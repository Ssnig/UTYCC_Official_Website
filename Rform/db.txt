    FOREIGN KEY (main_form_id) REFERENCES main_form(id) ON DELETE CASCADE
);

CREATE DATABASE IF NOT EXISTS `registration_db` CHARACTER SET utf8mb4 COLLATE utf8mb4_general_ci;
USE `registration_db`;

-- --------------------------------------------------------
-- Table: students
-- --------------------------------------------------------
CREATE TABLE `students` (
  `student_id` INT AUTO_INCREMENT PRIMARY KEY,
  `year` VARCHAR(20),
  `academic_year` VARCHAR(20),
  `seat_no` VARCHAR(50) UNIQUE,
  `entry_year` VARCHAR(10),
  `student_email` VARCHAR(100),
  `dormitory_preference` TINYINT(1) DEFAULT 0,
   `preferences` JSON,
  `student_name_mm` VARCHAR(100),
  `student_name_en` VARCHAR(100),
  `student_race` VARCHAR(50),
  `student_religion` VARCHAR(50),
  `student_nrc` VARCHAR(50),
  `student_dob` DATE,
  `student_birthplace` VARCHAR(100),
  `state_region` VARCHAR(100),
  `township` VARCHAR(100),
  `student_phone` VARCHAR(20),
  `student_address` TEXT,
   `created_at` TIMESTAMP DEFAULT CURRENT_TIMESTAMP
  `date` TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  `photo_url` VARCHAR(255)

) ENGINE=InnoDB;

-- --------------------------------------------------------
-- Table: fathers
-- --------------------------------------------------------
CREATE TABLE `fathers` (
  `father_id` INT AUTO_INCREMENT PRIMARY KEY,
  `student_id` INT,
  
  `father_name_mm` VARCHAR(100),
  `father_name_en` VARCHAR(100),
  `father_race` VARCHAR(50),
  `father_religion` VARCHAR(50),
  `father_nrc` VARCHAR(50),
  `father_phone` VARCHAR(20),
  `father_info` TEXT,
  `father_deceased_year` YEAR NULL,
  `father_address` TEXT,
  
  CONSTRAINT `fk_father_student`
    FOREIGN KEY (`student_id`) REFERENCES `students`(`student_id`)
    ON DELETE CASCADE
) ENGINE=InnoDB;

-- --------------------------------------------------------
-- Table: mothers
-- --------------------------------------------------------
CREATE TABLE `mothers` (
  `mother_id` INT AUTO_INCREMENT PRIMARY KEY,
  `student_id` INT,
  
  `mother_name_mm` VARCHAR(100),
  `mother_name_en` VARCHAR(100),
  `mother_race` VARCHAR(50),
  `mother_religion` VARCHAR(50),
  `mother_nrc` VARCHAR(50),
  `mother_phone` VARCHAR(20),
  `mother_info` TEXT,
  `mother_deceased_year` YEAR NULL,
  `mother_address` TEXT,
  
  CONSTRAINT `fk_mother_student`
    FOREIGN KEY (`student_id`) REFERENCES `students`(`student_id`)
    ON DELETE CASCADE
) ENGINE=InnoDB;




CREATE TABLE `preferences` (
  `preference_id` INT AUTO_INCREMENT PRIMARY KEY,
  `student_id` INT NOT NULL,
  `preference_order` TINYINT NOT NULL,  -- 1 to 5
  `major_code` VARCHAR(10) NOT NULL,
  `created_at` TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  CONSTRAINT `fk_preferences_student`
    FOREIGN KEY (`student_id`) REFERENCES `students`(`student_id`)
    ON DELETE CASCADE
) ENGINE=InnoDB;
